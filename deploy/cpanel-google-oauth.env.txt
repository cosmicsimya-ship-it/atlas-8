# cPanel Node.js — Google OAuth Environment Variables
# Application Manager → your app → Environment Variables
# Add each KEY = VALUE pair (no quotes). Restart the Node app after saving.

GOOGLE_CLIENT_ID=PASTE_CLIENT_ID.apps.googleusercontent.com
GOOGLE_CLIENT_SECRET=PASTE_CLIENT_SECRET
# Prefer oauth/callback — LiteSpeed/ModSecurity often blocks "/api/auth/google/*"
GOOGLE_REDIRECT_URI=https://cosmicsimya.com/api/auth/oauth/callback
FRONTEND_ORIGIN=https://cosmicsimya.com
ATLAS_SECURE_COOKIES=true
NODE_ENV=production
